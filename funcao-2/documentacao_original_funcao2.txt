--Pseudocódigo:
L1    função ( A[n][n] ):
L2        contador = 0;
L3        para i variando de 1 até n faça:
L4            para j variando de i até n faça:
L5                soma = A[i][j] + A[j][i];
L6                resto = soma mod 5; // -> verifica se a soma é múltiplo de 5
L7                se resto = 0 então
L8                    contador = contador + 1;
L9        retorne contador;

--Complexidade:
L1: 0 -> declaração da função
L2: 1
L3: n+1
L4: (n²+3n)/2
L5: (n²+n)/2
L6: (n²+n)/2
L7: (n²+n)/2
L8: (n²+n)/2
L9: 1

--Fórmula:
T(n)= (5n²+9n+6)/2

Big(O):
Big(O) = O(n²)
